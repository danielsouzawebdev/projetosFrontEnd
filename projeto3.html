Painel do Tempo

HTML CSS JS avançado Bootstrap

Digite o nome de uma cidade de verdade e veja a temperatura atual e a previsão dos próximos 5 dias — dados reais, de uma API pública, sem precisar criar conta nem chave de API.

Pré-requisitos

Os três projetos anteriores, e o módulo de JavaScript Avançado (Fetch API, async/await, classes). Usaremos a Open-Meteo, uma API meteorológica gratuita e sem necessidade de cadastro.

Passo 4 — A classe ServicoClima

Em vez de espalhar chamadas fetch soltas pelo código, vamos organizar tudo numa classe — a mesma ideia de encapsulamento vista no material de POO, agora em JavaScript:

script.js
class ServicoClima {
  async buscarCoordenadas(nomeCidade) {
    const url = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(nomeCidade)}&count=1&language=pt&format=json`;
    const resposta = await fetch(url);

    if (!resposta.ok) {
      throw new Error(`Erro ao buscar a cidade: ${resposta.status}`);
    }

    const dados = await resposta.json();

    if (!dados.results || dados.results.length === 0) {
      throw new Error(`Cidade "${nomeCidade}" não encontrada.`);
    }

    const { latitude, longitude, name, country } = dados.results[0];
    return { latitude, longitude, nome: name, pais: country };
  }

  async buscarPrevisao(latitude, longitude) {
    const url = `https://api.open-meteo.com/v1/forecast?latitude=${latitude}&longitude=${longitude}` +
      `&current=temperature_2m,relative_humidity_2m,weather_code,wind_speed_10m` +
      `&daily=weather_code,temperature_2m_max,temperature_2m_min` +
      `&forecast_days=5` +
      `&timezone=auto`;

    const resposta = await fetch(url);

    if (!resposta.ok) {
      throw new Error(`Erro ao buscar previsão: ${resposta.status}`);
    }

    return resposta.json();
  }
}

const servicoClima = new ServicoClima();

encodeURIComponent converte caracteres especiais (espaços, acentos) para o formato seguro de URL — sem isso, buscar "São Paulo" quebraria a requisição. A desestruturação { latitude, longitude, name, country } extrai só os campos que interessam da resposta, que normalmente vem com muito mais informação do que precisamos. E forecast_days=5 limita a previsão a 5 dias — sem ele, a API devolve 7 por padrão.

Passo 5 — Traduzindo o código do tempo

A API devolve o clima como um número (weather_code), seguindo um padrão da Organização Meteorológica Mundial. Precisamos traduzir isso para algo legível:

script.js
function descreverTempo(codigo) {
  const mapa = {
    0: { texto: "Céu limpo", emoji: "☀️" },
    1: { texto: "Poucas nuvens", emoji: "🌤️" },
    2: { texto: "Parcialmente nublado", emoji: "⛅" },
    3: { texto: "Nublado", emoji: "☁️" },
    45: { texto: "Névoa", emoji: "🌫️" },
    51: { texto: "Garoa leve", emoji: "🌦️" },
    61: { texto: "Chuva leve", emoji: "🌧️" },
    63: { texto: "Chuva moderada", emoji: "🌧️" },
    65: { texto: "Chuva forte", emoji: "🌧️" },
    71: { texto: "Neve leve", emoji: "🌨️" },
    80: { texto: "Pancadas de chuva", emoji: "🌦️" },
    95: { texto: "Tempestade", emoji: "⛈️" }
  };

  return mapa[codigo] ?? { texto: "Condição desconhecida", emoji: "❓" };
}

O operador ?? (nullish coalescing) retorna o valor da direita só quando o da esquerda é null ou undefined — aqui, cobre qualquer código que não esteja no mapa, evitando que a página quebre com um código inesperado.

Passo 6 — Buscar e renderizar o clima atual

script.js
const climaAtualEl = document.querySelector("#climaAtual");

function renderizarClimaAtual(cidade, dados) {
  const atual = dados.current;
  const tempo = descreverTempo(atual.weather_code);

  climaAtualEl.innerHTML = `
    <h2 class="h5">${cidade.nome}, ${cidade.pais}</h2>
    <div class="temperatura">${tempo.emoji} ${Math.round(atual.temperature_2m)}°C</div>
    <p class="mb-0 text-muted">${tempo.texto}</p>
    <p class="text-muted small">Umidade: ${atual.relative_humidity_2m}% · Vento: ${Math.round(atual.wind_speed_10m)} km/h</p>
  `;
  climaAtualEl.classList.remove("d-none");
}

Agora ligue a busca ao formulário:

script.js
const formBusca = document.querySelector("#formBusca");
const inputCidade = document.querySelector("#inputCidade");

formBusca.addEventListener("submit", async (evento) => {
  evento.preventDefault();
  await buscarEExibir(inputCidade.value);

async function buscarEExibir(nomeCidade) {
  const cidade = await servicoClima.buscarCoordenadas(nomeCidade);
  const dados = await servicoClima.buscarPrevisao(cidade.latitude, cidade.longitude);

  renderizarClimaAtual(cidade, dados);
}

Teste agora: digite "São Paulo" ou qualquer cidade grande e clique em Buscar. O card de clima atual deve aparecer com temperatura, descrição, umidade e vento reais.

Passo 7 — Previsão de 5 dias

script.js
const previsaoEl = document.querySelector("#previsao");

function renderizarPrevisao(dados) {
  const dias = dados.daily.time;
  previsaoEl.innerHTML = "";

  dias.forEach((diaISO, indice) => {
    const tempo = descreverTempo(dados.daily.weather_code[indice]);
    const max = Math.round(dados.daily.temperature_2m_max[indice]);
    const min = Math.round(dados.daily.temperature_2m_min[indice]);
    const data = new Date(diaISO + "T12:00:00");
    const diaSemana = data.toLocaleDateString("pt-BR", { weekday: "short" });

    previsaoEl.innerHTML += `
      <div class="col">
        <div class="card card-previsao p-2">
          <strong>${diaSemana}</strong>
          <div>${tempo.emoji}</div>
          <div>${max}° / ${min}°</div>
        </div>
      </div>
    `;
  });
}

E chame essa função dentro de buscarEExibir, logo depois de renderizarClimaAtual:

async function buscarEExibir(nomeCidade) {
  const cidade = await servicoClima.buscarCoordenadas(nomeCidade);
  const dados = await servicoClima.buscarPrevisao(cidade.latitude, cidade.longitude);

  renderizarClimaAtual(cidade, dados);
  renderizarPrevisao(dados);
}

A API devolve daily.time como um array de datas ("2026-10-07") e os outros valores (weather_code, temperaturas) como arrays paralelos — o valor no índice 2 de cada array sempre se refere ao mesmo dia. É por isso que o forEach usa o indice para buscar nos outros arrays.

Passo 8 — Loading e tratamento de erros

Buscas na internet levam tempo, e cidades digitadas errado vão acontecer. Reescreva buscarEExibir por completo:

script.js
const carregandoEl = document.querySelector("#carregando");
const erroEl = document.querySelector("#erro");

async function buscarEExibir(nomeCidade) {
  erroEl.classList.add("d-none");
  climaAtualEl.classList.add("d-none");
  previsaoEl.innerHTML = "";
  carregandoEl.classList.remove("d-none");

  try {
    const cidade = await servicoClima.buscarCoordenadas(nomeCidade);
    const dados = await servicoClima.buscarPrevisao(cidade.latitude, cidade.longitude);

    renderizarClimaAtual(cidade, dados);
    renderizarPrevisao(dados);
  } catch (erro) {
    erroEl.textContent = erro instanceof TypeError
      ? "Não foi possível conectar. Verifique sua internet e tente de novo."
      : erro.message;
    erroEl.classList.remove("d-none");
  } finally {
    carregandoEl.classList.add("d-none");
  }
}

O finally roda sempre — com sucesso ou com erro — por isso é o lugar certo para esconder o spinner de carregamento, em vez de repetir essa linha dentro do try e de novo dentro do catch. Já o instanceof TypeError distingue uma falha de rede (o fetch lança TypeError quando não consegue nem se conectar) de um erro que a sua própria classe lançou com uma mensagem já legível.

Teste o caminho de erro também

Digite uma cidade que não existe, tipo "Asdfgh123" — o alerta vermelho deve aparecer com a mensagem da sua própria classe ("Cidade ... não encontrada."), não um erro técnico confuso do navegador.

Passo 10 — Alternar modo escuro

script.js
const btnTema = document.querySelector("#btnTema");

btnTema.addEventListener("click", () => {
  const temaAtual = document.documentElement.getAttribute("data-bs-theme");
  const novoTema = temaAtual === "dark" ? "light" : "dark";

  document.documentElement.setAttribute("data-bs-theme", novoTema);
  btnTema.textContent = novoTema === "dark" ? "☀️ Modo claro" : "🌙 Modo escuro";

  localStorage.setItem("tema", novoTema);
});

// carregar o tema salvo, junto com a última cidade
const temaSalvo = localStorage.getItem("tema");
if (temaSalvo === "dark") {
  document.documentElement.setAttribute("data-bs-theme", "dark");
  btnTema.textContent = "☀️ Modo claro";
}

Todos os componentes do Bootstrap usados neste projeto (card, alert, spinner-border, input-group) já reagem sozinhos ao data-bs-theme — nenhuma linha extra de CSS é necessária para o modo escuro funcionar.

Checklist final

  • Buscar uma cidade real mostra temperatura, descrição, umidade e vento atuais
  • A previsão de 5 dias aparece em cards, com dia da semana, emoji e temperaturas máxima/mínima
  • O spinner aparece durante a busca e some depois, com sucesso ou erro
  • Buscar uma cidade inexistente mostra uma mensagem de erro clara, sem quebrar a página
  • Recarregar a página busca automaticamente a última cidade pesquisada
  • O botão de modo escuro alterna o tema e lembra a escolha ao recarregar

Desafio extra

  1. Adicione um botão "Usar minha localização", usando a Geolocation API do navegador (navigator.geolocation.getCurrentPosition) para buscar o clima sem digitar cidade nenhuma.
  2. Guarde as últimas 3 cidades buscadas (não só uma) e mostre como botões de atalho abaixo do formulário.
  3. Troque a previsão de 5 dias por um gráfico simples (usando <canvas> e a biblioteca Chart.js via CDN) mostrando a variação de temperatura.
